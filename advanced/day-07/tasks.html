1. Запись или определение
Легко7 мин

различать Set и CreateDataProperty по поведению.

У прототипа есть сеттер word и свойство id только для чтения. Пять способов дать объекту свойство — предскажите две строки.

const log = [];
const proto = { set word(v) { log.push("set " + v); } };
Object.defineProperty(proto, "id", { value: 0, writable: false });
const make = () => Object.create(proto);
const a = make(); a.word = "kitap";
const b = Object.assign(make(), { word: "alma" });
const c = { ...{ word: "galam" } };
const d = make(); d.id = 5;
const e = Object.defineProperty(make(), "id", { value: 7 });
console.log(Object.hasOwn(a, "word"), Object.hasOwn(b, "word"), Object.hasOwn(c, "word"), d.id, e.id, log.join(" | "));
try { Object.assign(make(), { id: 9 }); } catch (err) { console.log(err.name); }
  • предсказаны обе строки;
  • для каждого из пяти способов названа операция (Set или определение) и что сделал прототип;
  • объяснено, почему d.id = 5 молчит, а Object.assign бросает: где флажок Throw.

3. Шаги итератора
Легко6 мин

предсказывать вызовы next и return.

const log = [];
const make = (n) => ({
  [Symbol.iterator]() {
    let i = 0;
    log.push("iter");
    return {
      next() { i++; log.push("next"); return { value: i, done: i > n }; },
      return() { log.push("return"); return {}; },
    };
  },
});
for (const x of make(2)) log.push("body " + x);
console.log(log.join(" "));
log.length = 0;
for (const x of make(5)) { if (x === 2) break; }
console.log(log.join(" "));
log.length = 0;
const [p, q] = make(5);
console.log(p, q, log.join(" "));
log.length = 0;
const [r, ...rest] = make(2);
console.log(r, rest.join(), log.join(" "));
  • предсказаны четыре строки целиком;
  • для каждой отмечено, закончился итератор сам или его закрыли;
  • объяснено, почему [p, q] зовёт return, а [r, ...rest] — нет.

Основные задания

4. Свой CopyDataProperties
Средне12 мин

перенести 7.3.25 в код.

Напишите copyData(target, source, excluded = []) по шагам 7.3.25 и верните target. Ключи — Object.getOwnPropertyNames и getOwnPropertySymbols, свойство — Object.defineProperty с атрибутами CreateDataProperty.

const src = { b: 1, 2: "two", get g() { return "getter"; }, [Symbol.for("s")]: true };
Object.defineProperty(src, "hidden", { value: 1, enumerable: false });
const rest = copyData({}, src, ["b"]);
Object.keys(rest);                 // ["2", "g"]
rest.g;                            // "getter"
rest[Symbol.for("s")];             // true
Object.keys(copyData({}, "ab"));   // ["0", "1"]
copyData({}, null);                // {}
  • все пять строк примера дают то, что в комментариях;
  • ключи совпадают с const { b, ...real } = src;
  • на получателе с сеттером g показано: после copyData у него свойство-значение, а Object.assign позвал бы сеттер.
5. Свой GroupBy
Средне12 мин

перенести 7.3.34–7.3.35 и обе обёртки в код.

Напишите groupBy(items, callback, keyCoercion) по шагам 7.3.35 — список групп { key, elements }; ключ "property" — ToPropertyKey, "collection" — −0 в +0; группа — по Object.is. Над ней — groupObject и groupMap. Перебор через for...of сам закроет итератор, если колбэк бросит.

const words = ["kitap", "alma", "galam", "at"];
JSON.stringify(groupObject(words, (w) => w.length));   // '{"2":["at"],"4":["alma"],"5":["kitap","galam"]}'
Object.getPrototypeOf(groupObject([], (x) => x));     // null
groupMap([1, "1", -0, 0], (x) => x).size;             // 3
groupMap([-0], (x) => x).has(0);                      // true
  • все четыре строки примера дают то, что в комментариях;
  • groupObject(null, f) и groupObject([], 5) бросают TypeError до перебора — как Object.groupBy;
  • объяснено, почему в JSON ключи идут 2, 4, 5, хотя группы появились в порядке 5, 4, 2.
6. «Карточки 2.0»: настройки из localStorage
Средне14 мин

сливать сохранённые данные с настройками по умолчанию без чужих ключей.

Добавим в «Карточки 2.0» настройки — тему и число карточек в день. Они лежат в localStorage строкой JSON, испортить которую мог кто угодно. Напишите mergeSettings(defaults, text): новый объект, где ключи defaults взяты из сохранённого, если они там есть; испорченный JSON или не объект — просто defaults; ни лишних ключей, ни смены прототипа.

const defaults = { theme: "light", perDay: 20 };
mergeSettings(defaults, '{"perDay": 30}');                         // { theme: "light", perDay: 30 }
mergeSettings(defaults, '{"__proto__": {"admin": true}}').admin;   // undefined
Object.getPrototypeOf(mergeSettings(defaults, '{"__proto__": {}}')) === Object.prototype;   // true
mergeSettings(defaults, "не JSON");                                // { theme: "light", perDay: 20 }
mergeSettings(defaults, "null");                                   // { theme: "light", perDay: 20 }
mergeSettings(defaults, '{"perDay": 30, "extra": 1}').extra;       // undefined
  • все шесть строк примера дают то, что в комментариях;
  • показано, на каких строках ошибается Object.assign({}, defaults, JSON.parse(text)), и названа операция, которая виновата;
  • defaults после всех вызовов не изменился.

8. Свой for...of
Сложно10 мин

собрать перебор из GetIterator (7.4.4), шагов 14.7.5.7 и IteratorClose (7.4.11).

Напишите forOf(iterable, body) без for...of, spread и деструктуризации; body, вернув "break", останавливает перебор. next читается один раз, результат проверяется на «объект», done — ToBoolean.

const log = [];
forOf([1, 2, 3], (x) => { log.push(x); if (x === 2) return "break"; });
const closing = { [Symbol.iterator]() { let i = 0; return { next: () => ({ value: ++i, done: i > 3 }), return: () => { log.push("return"); return {}; } }; } };
forOf(closing, (x) => { log.push(x); if (x === 2) return "break"; });
forOf(closing, (x) => { log.push(x); });
log.join();   // "1,2,1,2,return,1,2,3"
  • строка примера даёт то, что в комментарии;
  • ошибки как у for...of: forOf(5, f) — TypeError; ошибка body важнее ошибки return; return, вернувший 5, после "break" — TypeError;
  • объяснено, почему при естественном конце return не зовут.
9. Незнакомый раздел: сеттер 7.3.37
Сложно10 мин

прочитать операцию, которая на лекции была строкой справочника.

constructor и Symbol.toStringTag у Iterator.prototype (день 13 курса 2) — аксессоры с сеттером 7.3.37 SetterThatIgnoresPrototypeProperties. Прочитайте четыре шага и предскажите пять строк.

const attempt = (f) => { try { f(); return "ok"; } catch (e) { return e.name; } };
const it = [1, 2].values();
const plain = Object.create(Iterator.prototype);
const desc = Object.getOwnPropertyDescriptor(Iterator.prototype, "constructor");
console.log(typeof desc.get, typeof desc.set, "value" in desc);
console.log(attempt(() => { Iterator.prototype.constructor = 5; }), attempt(() => { it.constructor = 5; }), Object.hasOwn(it, "constructor"), it.constructor);
console.log(attempt(() => { plain[Symbol.toStringTag] = "Моё"; }), String(plain), String([3].values()));
console.log(attempt(() => desc.set.call(7, 1)), attempt(() => desc.set.call(Iterator.prototype, 1)));
const fixed = Object.defineProperty(Object.create(Iterator.prototype), "constructor", { value: 1, writable: false, configurable: true });
console.log(attempt(() => desc.set.call(fixed, 2)), fixed.constructor);
  • предсказаны пять строк, у каждого ответа — номер шага 7.3.37;
  • по примечанию к шагу 2 объяснено, почему присваивание Iterator.prototype.constructor = 5 бросает даже в нестрогом скрипте;
  • объяснено, почему fixed.constructor = 2 до этого сеттера не дошло бы.
