const frame = document.createElement("iframe");

2. Откуда this
Легко7 мин

проходить GetThisEnvironment и GetNewTarget по шагам.

const who = (v) => (v === undefined ? "undefined" : v === globalThis ? "global" : v.name);
const obj = {
  name: "obj",
  arrow: () => who(this),
  method() {
    const inner = function () { return who(this); };
    return [who(this), (() => who(this))(), inner(), inner.call({ name: "call" })].join(" ");
  },
  nested() { return { name: "inner", get: () => who(this) }.get(); },
};
console.log(who(this), obj.arrow(), obj.method(), obj.nested());
function F() { console.log((() => (new.target ? new.target.name : "нет"))()); }
F();
new F();
  • предсказаны девять значений, для каждой стрелки названа запись окружения, где кончился поиск;
  • объяснено, почему nested даёт «obj», а не «inner»;
  • для inner() назван шаг OrdinaryCallBindThis (10.2.1.2), который дал глобальный this.

Основные задания

4. Своя GetThisEnvironment
Средне12 мин

перенести 9.4.3–9.4.5 и HasThisBinding записей в код.

Запись окружения — объект: { kind: "declarative", outer }, { kind: "function", status, thisValue, newTarget, outer } (status — "lexical", "initialized" или "uninitialized"), { kind: "module", outer }, { kind: "global", thisValue, outer: null }. Напишите hasThisBinding(env) — кусками по видам, getThisEnvironment(env), resolveThisBinding(env) (ReferenceError, пока this не связан) и getNewTarget(env).

const attempt = (f) => { try { return f(); } catch (e) { return e.name; } };
const global = { kind: "global", thisValue: "window", outer: null };
const method = { kind: "function", status: "initialized", thisValue: "obj", newTarget: undefined, outer: global };
const arrow = { kind: "function", status: "lexical", outer: method };
const block = { kind: "declarative", outer: arrow };
const ctor = { kind: "function", status: "uninitialized", newTarget: "Child", outer: global };
const inCtor = { kind: "function", status: "lexical", outer: ctor };
resolveThisBinding(block);                                           // "obj"
getThisEnvironment(block) === method;                                // true
resolveThisBinding({ kind: "declarative", outer: { kind: "module", outer: global } });   // undefined
getNewTarget(inCtor);                                                // "Child"
attempt(() => resolveThisBinding(inCtor));                           // "ReferenceError"
  • все пять строк примера дают то, что в комментариях;
  • поиск — один цикл по outer, как шаг 2 раздела 9.4.3, без проверки на null, и объяснено, почему она не нужна;
  • та же цепочка в настоящем коде — блок в стрелке в методе — даёт тот же ответ.
5. Своя очередь заданий
Средне14 мин

собрать требования 9.5 и порядок браузера в одной модели.

Напишите makeLoop() с методами script(fn) и task(fn) — поставить задачу, microtask(fn), log(text), run() и journal(). run берёт задачи по одной; после каждой выполняет все микрозадачи, даже добавленные по ходу. Ошибка задания не останавливает очередь: в журнал пишется «ошибка: текст».

const loop = makeLoop();
loop.script(() => {
  loop.log("скрипт");
  loop.task(() => loop.log("таймер"));
  loop.microtask(() => {
    loop.log("микро 1");
    loop.microtask(() => loop.log("микро 3"));
  });
  loop.microtask(() => loop.log("микро 2"));
});
loop.script(() => {
  loop.microtask(() => loop.log("микро второго скрипта"));
  throw new Error("сбой");
});
loop.run();
loop.journal().join(" | ");   // "скрипт | микро 1 | микро 2 | микро 3 | ошибка: сбой | микро второго скрипта | таймер"
  • строка примера даёт то, что в комментарии;
  • первый скрипт, переписанный на setTimeout и Promise.resolve().then, печатает в Chrome тот же порядок;
  • каждое требование 9.5 (пустой стек, одно за раз, до конца, свои ошибки) показано строкой модели.
6. «Карточки 2.0»: одна загрузка на всех
Средне12 мин

не начинать работу второй раз, пока идёт первая.

Две буквы подряд грузят колоды дважды (раздел 16 лекции). Напишите once(asyncFn): пока вызов не закончился, все вызовы получают тот же промис; после выполнения или отказа следующий вызов начинает заново. Оберните ею openAll в своём main.js.

let loads = 0;
const slow = async () => {
  loads++;
  await null;
  return "колоды";
};
const openAll = once(slow);
const first = openAll();
first === openAll();   // true
loads;                 // 1
  • обе строки примера дают то, что в комментариях;
  • после того как first выполнился, новый вызов грузит снова: loads — 2;
  • после отказа следующий вызов тоже пробует снова, а не отдаёт старый отказ.

Со звёздочкой

8. Свои ResolveBinding и PutValue
Сложно10 мин

пройти путь имени: 9.4.2, 9.1.2.1, 6.2.5.5–6.2.5.6.

Запись окружения — { bindings: Map, outer }. Напишите resolveBinding(name, env, strict) — ссылку { base, name, strict }, где base — запись с именем или "unresolvable"; getValue(ref); putValue(ref, value, globalObject) — globalObject играет роль глобального объекта текущего реалма.

const attempt = (f) => { try { return f(); } catch (e) { return e.name; } };
const globalObject = {};
const global = { bindings: new Map([["label", "страница"]]), outer: null };
const fn = { bindings: new Map([["x", 1]]), outer: global };
getValue(resolveBinding("x", fn, false));                                     // 1
getValue(resolveBinding("label", fn, false));                                 // "страница"
putValue(resolveBinding("leak", fn, false), 5, globalObject);
globalObject.leak;                                                            // 5
attempt(() => putValue(resolveBinding("y", fn, true), 1, globalObject));     // "ReferenceError"
attempt(() => getValue(resolveBinding("nope", fn, false)));                  // "ReferenceError"
  • все пять проверяемых строк примера дают то, что в комментариях;
  • флажок strict хранится в ссылке, как [[Strict]] Reference Record (день 4), и читается только при записи;
  • объяснено, почему чтение неразрешимой ссылки бросает и в нестрогом коде.
9. Задание для thenable по шагам
Сложно10 мин

прочитать, как resolve ставит задание и какой реалм ему даёт.

Прочитайте 27.2.1.3 CreateResolvingFunctions и 27.2.2.2 NewPromiseResolveThenableJob, найдите вызов HostEnqueuePromiseJob и предскажите строку.

const log = [];
const thenable = { then(resolve) { log.push("then"); resolve("готово"); } };
Promise.resolve().then(() => log.push("1"));
new Promise((resolve) => resolve(thenable)).then((v) => log.push(v));
Promise.resolve().then(() => log.push("2")).then(() => log.push("3")).then(() => log.push("4"));
setTimeout(() => console.log(log.join(" ")), 0);
  • предсказана строка, у каждого слова — задание, которое его записало;
  • найдено, откуда берётся реалм задания: GetFunctionRealm функции then, и что будет, если это отозванный прокси (день 16);
  • объяснено, почему «готово» записано после «2»: сколько лишних заданий дал thenable и сколько дал бы вместо него выполненный промис (день 28 курса 2).
