const run = (f) => { try { f(); return "ok"; } catch (e) { return e.name; } };
const o = {};
Object.defineProperty(o, "a", { value: 1, writable: true, enumerable: true });

Основные задания

4. Своя проверка дескриптора
Средне14 мин

перенести 10.1.6.3 ValidateAndApplyPropertyDescriptor в код.

Напишите validate(current, desc, extensible) → true или false, как ValidateAndApplyPropertyDescriptor без объекта (IsCompatiblePropertyDescriptor, 10.1.6.2): current — полный дескриптор или undefined, desc — частичный. Аксессор — есть get или set, данные — есть value или writable; сравнение — Object.is (SameValue).

const data = (v, w, e, c) => ({ value: v, writable: w, enumerable: e, configurable: c });
validate(undefined, { value: 1 }, true);                          // true
validate(undefined, { value: 1 }, false);                         // false
validate(data(1, true, true, false), { value: 2 }, true);         // true
validate(data(1, false, true, false), { value: 2 }, true);        // false
validate(data(NaN, false, true, false), { value: NaN }, true);    // true
validate(data(0, false, true, false), { value: -0 }, true);       // false
validate(data(1, true, true, false), { get() {} }, true);         // false
validate(data(1, true, true, true), { get() {} }, true);          // true
validate(data(1, true, true, false), {}, true);                   // true
  • все девять строк примера дают то, что в комментариях;
  • те же девять случаев через Object.defineProperty настоящего объекта: true — без ошибки, false — TypeError;
  • объяснено, зачем спецификации вариант без объекта (подсказка: инварианты Proxy, день 16).
5. Свой OrdinarySet
Средне12 мин

перенести 10.1.9.2 в код с Receiver.

Напишите ordinarySet(O, P, V, Receiver) → true или false по шагам раздела 6 лекции, без присваивания: свойства читайте Object.getOwnPropertyDescriptor и Object.getPrototypeOf, пишите Object.defineProperty, сеттер зовите call. Нерасширяемый Receiver — false.

const proto = {};
Object.defineProperty(proto, "ro", { value: 1, writable: false });
Object.defineProperty(proto, "acc", { set(v) { this.log = v; } });
proto.rw = 0;
const a = Object.create(proto);
ordinarySet(a, "rw", 5, a);                     // true
[a.rw, proto.rw];                               // [5, 0]
ordinarySet(a, "ro", 5, a);                     // false
ordinarySet(a, "acc", 7, a);                    // true
[a.log, Object.hasOwn(a, "acc")];               // [7, false]
ordinarySet(a, "nope", 1, "строка");            // false
  • все строки примера дают то, что в комментариях;
  • на тех же объектах настоящая запись в строгом коде совпадает с моделью: true — без ошибки, false — TypeError;
  • случай super.x = раздела 6: ordinarySet(proto, "rw", 9, obj), где у obj свой аксессор rw, — false; объяснено, какой шаг его дал.
6. Своя OrdinaryOwnPropertyKeys
Средне10 мин

записать порядок ключей как алгоритм.

Ключи даны в порядке создания. Напишите ownKeysOrder(keys): индексы массива по возрастанию числа, потом остальные строки, потом символы — по порядку создания. Индекс — строка, которая равна String(Number(s)) и даёт целое от 0 до 2³² − 2.

const s = Symbol("s");
ownKeysOrder(["b", "10", s, "a", "2", "01", "4294967295", "4294967294"]).map(String);   // ["2", "10", "4294967294", "b", "a", "01", "4294967295", "Symbol(s)"]
ownKeysOrder(["-0", "0", "1.5", "1e3", "7"]);                                           // ["0", "7", "-0", "1.5", "1e3"]
  • обе строки примера дают то, что в комментариях;
  • те же ключи, созданные у настоящего объекта в том же порядке, дают в getOwnPropertyNames и getOwnPropertySymbols тот же порядок;
  • объяснено, почему "-0" и "1e3" — не индексы.
7. «Карточки 2.0»: заморозить и не сломать наследника
Средне14 мин

выбрать момент заморозки по [[Construct]].

В разделе 17 лекции Object.freeze(this) в конструкторе Card ломал поле наследника. Перепишите Card и BigCard extends Card { level = 2; } так, чтобы карточку любого из двух классов нельзя было изменить снаружи, а поле level работало. Подсказка: new.target говорит, какой класс вызвали с new.

const run = (f) => { try { f(); return "ok"; } catch (e) { return e.name; } };
const card = new Card("kitap");
const big = new BigCard("suw");
run(() => Object.defineProperty(card, "word", { value: "x" }));                     // "TypeError"
[big.word, big.level, Object.isFrozen(big), Object.isFrozen(card)];                 // ["suw", 2, true, true]
  • обе строки примера дают то, что в комментариях;
  • class HugeCard extends BigCard { size = 3; } создаётся без ошибки, но не заморожен — объяснено, почему, и что в него дописать;
  • названы шаги [[Construct]] и super(), которые решают, когда поля наследника уже стоят.

Со звёздочкой

8. Свой [[Construct]] для обычной функции
Сложно12 мин

повторить 10.2.2 для base без new.

Напишите construct(F, args, newTarget = F): прототип — newTarget.prototype, а если он не объект — Object.prototype (GetPrototypeFromConstructor); вызвать F с этим this; вернули объект — он и ответ, иначе — созданный объект.

function Point(x) { this.x = x; }
function Swap() { this.a = 1; return { swapped: true }; }
function Prim() { this.a = 1; return 42; }
function Other() {}
construct(Point, [3]).x;                                                 // 3
Object.getPrototypeOf(construct(Point, [3])) === Point.prototype;        // true
[construct(Swap, []).swapped, construct(Prim, []).a];                    // [true, 1]
Object.getPrototypeOf(construct(Point, [1], Other)) === Other.prototype; // true
  • все строки примера дают то, что в комментариях;
  • при Point.prototype = null модель и настоящий new Point(1) дают Object.prototype;
  • названы три вещи [[Construct]], которых в модели нет: поля, derived и реалм конструктора.

const parse = (code) => { try { new Function(code)(); return "ok"; } catch (e) { return e.name; } };
console.log(["class A { static prototype() {} }", "class A { static ['proto' + 'type']() {} }", "class A { static ['proto' + 'type'] = 1; }",
