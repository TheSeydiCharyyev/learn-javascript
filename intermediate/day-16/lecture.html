1. Проект вместо новой темы

Сегодня новых методов не будет. Пятнадцать дней курса дали всё, чтобы собрать приложение иначе, чем на дне 29 Essentials. Там данные были массивом простых объектов, и любая функция могла испортить любое поле. Сегодня данные — объекты своих классов: снаружи видно только то, что класс разрешил, а неверную оценку класс просто не примет. Собираем журнал группы: ученики, их оценки, средний балл и темы занятий.

3. Настройки под замком

Первое в app.js — строгий режим дня 3: тихие ошибки станут громкими. Затем — постоянные данные программы: название группы, допустимые оценки и порог отличника. Их никто не должен менять, поэтому — deepFreeze из дня 6:

"use strict";

function deepFreeze(obj) {
  Object.freeze(obj);
  for (const value of Object.values(obj)) {
    if (typeof value === "object" && value !== null && !Object.isFrozen(value)) deepFreeze(value);
  }
  return obj;
}

const SETTINGS = deepFreeze({
  title: "JS-1",
  marks: [2, 3, 4, 5],
  excellent: 4.5,
});

Обычный Object.freeze здесь не хватило бы: массив marks — отдельный объект, и неглубокая заморозка его не трогает. Проверка шага в консоли:

console.log(Object.isFrozen(SETTINGS), Object.isFrozen(SETTINGS.marks), SETTINGS.marks.includes(5));   // true true true
SETTINGS.marks.push(6);
// TypeError: Cannot add property 4, object is not extensible

push падает в любом режиме — это помнит день 6. А присваивание SETTINGS.excellent = 4 в консоли просто не сработает, а внутри app.js со строгим режимом упадёт с Cannot assign to read only property.

4. Класс Student

Ученик — имя и оценки. Номер он получает сам из счётчика класса, а оценки хранит в приватном поле, куда снаружи не дотянуться:

class Student {
  static #count = 0;
  #id;
  #name;
  #marks = [];

  constructor(name) {
    Student.#count++;
    this.#id = Student.#count;
    this.#name = name;
  }

  get id() {
    return this.#id;
  }

  get name() {
    return this.#name;
  }

  get marks() {
    return [...this.#marks];
  }

  get average() {
    if (this.#marks.length === 0) return null;
    return this.#marks.reduce((a, b) => a + b) / this.#marks.length;
  }

  get isExcellent() {
    return this.average !== null && this.average >= SETTINGS.excellent;
  }

  addMark(mark) {
    if (!SETTINGS.marks.includes(mark)) {
      throw new RangeError(`Оценка ${mark}: можно только ${SETTINGS.marks.join(", ")}`);
    }
    this.#marks.push(mark);
    return this;
  }
}

Почти каждая строка — знакомый приём. Счётчик static #count общий для класса и спрятан (день 9), как nextId дня 29 Essentials, только теперь его не изменить снаружи. Геттеры id и name без сеттеров — «только для чтения». marks отдаёт копию массива; почему это важно — раздел 13. average и isExcellent вычисляются при каждом чтении, поэтому всегда верны (день 5). Ученик без оценок получает средний null, а не NaN. Единственный способ добавить оценку — addMark, и он проверяет её по настройкам. return this даёт цепочку вызовов (день 8).

const gulnar = new Student("Гүлнар");
gulnar.addMark(5).addMark(4);
console.log(gulnar.id, gulnar.name, gulnar.marks, gulnar.average, gulnar.isExcellent);   // 1 Гүлнар [5, 4] 4.5 true
gulnar.marks.push(2);
console.log(gulnar.marks.length);   // 2 — изменилась только копия
gulnar.addMark(7);
// RangeError: Оценка 7: можно только 2, 3, 4, 5

Средний 4,5 — уже отличник: условие >=. Попытка дописать оценку через копию ничего не изменила, а неверная оценка остановлена ошибкой.

5. Класс Group

Группа хранит учеников в Map по номеру (день 14) — «быстро найти ученика по номеру», как обещал день 36 Essentials. Темы — в Set (день 15): тема бывает в группе один раз.

class Group {
  #title;
  #students = new Map();
  #topics = new Set();

  constructor(title) {
    this.#title = title;
  }

  get title() {
    return this.#title;
  }

  get size() {
    return this.#students.size;
  }

  get topics() {
    return [...this.#topics];
  }

  get average() {
    const averages = Iterator.from(this).map((s) => s.average).filter((a) => a !== null).toArray();
    if (averages.length === 0) return null;
    return averages.reduce((a, b) => a + b) / averages.length;
  }

  *[Symbol.iterator]() {
    yield* this.#students.values();
  }

  add(name) {
    const clean = name.trim();
    if (clean === "") throw new Error("Напишите имя ученика");
    if (Iterator.from(this).some((s) => s.name === clean)) throw new Error(`${clean} уже в группе`);
    const student = new Student(clean);
    this.#students.set(student.id, student);
    return student;
  }

  get(id) {
    return this.#students.get(id);
  }

  remove(id) {
    return this.#students.delete(id);
  }

  addTopic(topic) {
    const clean = topic.trim();
    if (clean === "") throw new Error("Напишите тему");
    if (this.#topics.has(clean)) throw new Error(`Тема «${clean}» уже есть`);
    this.#topics.add(clean);
  }
}

Все проверки ввода — здесь, в классе: пустое имя, повтор, пустая и повторная тема. Метод бросает ошибку с готовым текстом для человека, а показывать её — дело страницы. Set и сам не пустил бы вторую «Map», но молча; человеку нужно сообщение, поэтому сначала has. Средний по группе считают только ученики с оценками.

И данные журнала — сразу за классами:

const group = new Group(SETTINGS.title);
group.add("Аман").addMark(5).addMark(4).addMark(5);
group.add("Мерет").addMark(3).addMark(4);
group.add("Айна").addMark(5).addMark(5);
["классы", "Map", "Set"].forEach((topic) => group.addTopic(topic));
let filter = "all";

add возвращает нового ученика, поэтому оценки ставятся тут же цепочкой. Переменная filter — что показывать: "all" или "excellent". Проверка шага:

console.log(group.size, group.get(2).name, group.get(2).average, group.topics);   // 3 Мерет 3.5 ['классы', 'Map', 'Set']
console.log(group.average.toFixed(2), group.get("2"));                           // 4.39 undefined
group.add(" Аман ");
// Error: Аман уже в группе

group.get("2") — undefined: ключи Map не превращаются в строки, и строка "2" — другой ключ (день 14). Это ещё встретится.

6. Свой итератор

Строка *[Symbol.iterator]() делает группу перебираемой (день 11), а генератор (день 12) пишет протокол за нас. yield* отдаёт по одному всё, что выдаёт итератор this.#students.values(), — учеников в порядке добавления.

for (const student of group) console.log(student.name, student.marks.join(" "));

console.log(String(group[Symbol.iterator]()));

Мерет 3 4

Снаружи никто не знает, что внутри Map: группа просто перебирается, как массив. Методы итераторов дня 13 работают через Iterator.from(group). 8. Главная схема

Схема дня 29 остаётся: страница — картинка данных, слушатель меняет данные и зовёт render. Новое — кто меняет данные. Раньше слушатель сам писал в массив и сам проверял ввод. Теперь он только зовёт метод класса, а метод либо меняет данные, либо бросает ошибку с текстом для человека.

Выгода: проверку нельзя забыть. Добавит ученика форма, консоль или будущий импорт из файла — путь один, через group.add, и пустое имя не пройдёт нигде. На дне 29 проверка жила в слушателе формы, и tasks.push из консоли её обходил.

9. Один ученик — одна строка

function format(average) {
  return average === null ? "—" : average.toFixed(2);
}

function makeRow(student) {
  const tr = document.createElement("tr");
  const name = document.createElement("td");
  const marks = document.createElement("td");
  const average = document.createElement("td");
  const actions = document.createElement("td");

  name.textContent = student.name;
  marks.textContent = student.marks.join(", ");
  average.textContent = format(student.average);
  for (const mark of SETTINGS.marks) {
    const button = document.createElement("button");
    button.textContent = mark;
    button.dataset.mark = mark;
    actions.append(button, " ");
  }
  const del = document.createElement("button");
  del.textContent = "✕";
  del.classList.add("delete");
  actions.append(del);

  tr.append(name, marks, average, actions);
  tr.dataset.id = student.id;
  tr.classList.toggle("excellent", student.isExcellent);
  return tr;
}

Это makeItem дня 29: функция создаёт строку и ничего не вставляет. Читает она только геттеры — до приватных полей ей не дотянуться, и это правильно. Кнопки оценок строятся из настроек: появится оценка 1 в SETTINGS.marks — появится и кнопка. data-id — ниточка от строки к ученику, как раньше.

const row = makeRow(group.get(1));
console.log(row.dataset.id, row.className, row.children[1].textContent, row.children[2].textContent, row.querySelectorAll("button").length);
// 1 excellent 5, 4, 5 4.67 5

10. render: страница из группы

function isVisible(student) {
  return filter === "all" || student.isExcellent;
}

function render() {
  title.textContent = `Журнал группы ${group.title}`;
  rows.textContent = "";
  rows.append(...Iterator.from(group).filter(isVisible).map(makeRow));

  const excellent = Iterator.from(group).filter((s) => s.isExcellent).toArray().length;
  summary.textContent = `Учеников: ${group.size} · средний по группе: ${format(group.average)} · отличников: ${excellent}`;
  topicsLine.textContent = `Темы (${group.topics.length}): ${group.topics.join(", ")}`;
  filters.querySelectorAll("button").forEach((b) => b.classList.toggle("active", b.dataset.filter === filter));
}

render();

Строка rows.append(...) — сердце программы. На дне 29 было tasks.filter(isVisible).map(makeItem) по массиву; теперь та же цепочка идёт по итератору группы, и никакого массива не строится: ученик проходит фильтр, превращается в строку и уходит в append (день 13). Итоги считаются по всей группе, а не по видимым строкам.

11. Две формы, одна проверка

У обеих форм одинаковая работа: взять текст из поля, отдать его методу группы и либо показать ошибку, либо очистить форму и перерисовать. Одинаковую работу пишут один раз:

function submitWith(form, action) {
  const input = form.querySelector("input");
  form.addEventListener("submit", (event) => {
    event.preventDefault();
    try {
      action(input.value);
    } catch (e) {
      error.textContent = e.message;
      input.focus();
      return;
    }
    error.textContent = "";
    form.reset();
    render();
  });
}

submitWith(studentForm, (name) => group.add(name));
submitWith(topicForm, (topic) => group.addTopic(topic));

try...catch дня 11 Essentials ловит ошибку, которую бросил класс, и показывает её message. Слушатель не знает ни одного правила про имена и темы — все они в классе. Обратите внимание на стрелку (name) => group.add(name): без неё метод потеряет this (раздел 14).

Проверка шага: наберите «  Сапар  », нажмите Enter, потом «Сапар» ещё раз.

console.log(group.size, group.get(4).name, rows.children.length, error.textContent);   // 4 Сапар 4 Сапар уже в группе

12. Оценки, удаление, фильтр

function studentOf(element) {
  return group.get(Number(element.closest("tr").dataset.id));
}

rows.addEventListener("click", (event) => {
  const button = event.target.closest("button");
  if (!button) return;
  const student = studentOf(button);
  if (button.matches(".delete")) {
    group.remove(student.id);
  } else {
    student.addMark(Number(button.dataset.mark));
  }
  render();
});

filters.addEventListener("click", (event) => {
  const button = event.target.closest("button");
  if (!button) return;
  filter = button.dataset.filter;
  render();
});

Один слушатель на все кнопки таблицы — делегирование дня 27 Essentials. studentOf — это taskOf дня 29, но ищет не перебором find, а по ключу Map. Здесь нет try...catch: кнопки дают только допустимые оценки. Если addMark всё-таки бросит ошибку, это ошибка программиста, и пусть она громко появится в консоли, а не тихо под формой.

Проверка шага на свежей странице: «3» у Айны, «✕» у Мерета, потом «Отличники».

console.log(group.get(3).marks, group.get(3).average.toFixed(2), group.size, rows.children.length);   // [5, 5, 3] 4.33 2 1
console.log(summary.textContent);   // Учеников: 2 · средний по группе: 4.50 · отличников: 1

Айна с тройкой опустилась до 4,33 и перестала быть отличницей, поэтому в списке «Отличники» остался только Аман. А средний по группе — 4,50: (4,67 + 4,33) / 2.

13. Разбор: средний больше пяти

Частая первая версия геттера отдаёт сам массив: get marks() { return this.#marks; }. Пока массив только читают, всё работает. Но однажды кто-то пишет в консоли или в другой функции:

group.get(1).marks.push(9);
console.log(group.get(1).marks, group.get(1).average);   // [5, 4, 5, 9] 5.75

Шаг 1. Что не так? Средний больше пяти — такой оценки быть не может, addMark не пропустил бы девятку.

Шаг 2. Как она попала? Мимо addMark. Приватное поле прячет переменную, но геттер отдал наружу ссылку на тот же массив, и push изменил его изнутри. Та же история, что с неглубокой заморозкой дня 6: защита стоит на объекте, а не на том, на что он ссылается.

Шаг 3. Чиним. Геттер отдаёт копию: return [...this.#marks];. Теперь push меняет копию, и раздел 4 это проверил: длина осталась 2.

14. Разбор: ещё три ошибки

Под формой — «Iterator.from called on non-object»

Хотели короче: submitWith(studentForm, group.add). Метод передан без объекта, и при вызове action(input.value) у него нет this — день 2. Тело класса строгое (день 8), поэтому this — undefined, и первая же строка, где он нужен, падает. catch честно показал текст ошибки человеку:

["Сапар", "Гүлнар"].forEach(group.add);
// TypeError: Iterator.from called on non-object

Странный текст — подсказка: ошибка не в данных, а в вызове. Лечится стрелкой (name) => group.add(name) или group.add.bind(group).

Нажимаю «5» — ничего

В слушателе забыли Number: student.addMark(button.dataset.mark). На странице ничего не меняется, а в консоли:

group.get(1).addMark("5");
// RangeError: Оценка 5: можно только 2, 3, 4, 5

Текст сбивает с толку: «пять нельзя, можно пять?» Всё, что читается из dataset, — строки (день 25 Essentials), а includes сравнивает по SameValueZero: "5" не равно 5. С Map то же самое: без Number в studentOf group.get("1") вернёт undefined (раздел 5), и упадёт уже Cannot read properties of undefined (reading 'addMark').

// app.js — журнал группы
"use strict";

// 1. Настройки — под замком
function deepFreeze(obj) {
  Object.freeze(obj);
  for (const value of Object.values(obj)) {
    if (typeof value === "object" && value !== null && !Object.isFrozen(value)) deepFreeze(value);
  }
  return obj;
}

const SETTINGS = deepFreeze({
  title: "JS-1",
  marks: [2, 3, 4, 5],
  excellent: 4.5,
});

// 2. Классы: данные и правила
class Student {
  static #count = 0;
  #id;
  #name;
  #marks = [];

  constructor(name) {
    Student.#count++;
    this.#id = Student.#count;
    this.#name = name;
  }

  get id() {
    return this.#id;
  }

  get name() {
    return this.#name;
  }

  get marks() {
    return [...this.#marks];
  }

  get average() {
    if (this.#marks.length === 0) return null;
    return this.#marks.reduce((a, b) => a + b) / this.#marks.length;
  }

  get isExcellent() {
    return this.average !== null && this.average >= SETTINGS.excellent;
  }

  addMark(mark) {
    if (!SETTINGS.marks.includes(mark)) {
      throw new RangeError(`Оценка ${mark}: можно только ${SETTINGS.marks.join(", ")}`);
    }
    this.#marks.push(mark);
    return this;
  }
}

class Group {
  #title;
  #students = new Map();
  #topics = new Set();

  constructor(title) {
    this.#title = title;
  }

  get title() {
    return this.#title;
  }

  get size() {
    return this.#students.size;
  }

  get topics() {
    return [...this.#topics];
  }

  get average() {
    const averages = Iterator.from(this).map((s) => s.average).filter((a) => a !== null).toArray();
    if (averages.length === 0) return null;
    return averages.reduce((a, b) => a + b) / averages.length;
  }

  *[Symbol.iterator]() {
    yield* this.#students.values();
  }

  add(name) {
    const clean = name.trim();
    if (clean === "") throw new Error("Напишите имя ученика");
    if (Iterator.from(this).some((s) => s.name === clean)) throw new Error(`${clean} уже в группе`);
    const student = new Student(clean);
    this.#students.set(student.id, student);
    return student;
  }

  get(id) {
    return this.#students.get(id);
  }

  remove(id) {
    return this.#students.delete(id);
  }

  addTopic(topic) {
    const clean = topic.trim();
    if (clean === "") throw new Error("Напишите тему");
    if (this.#topics.has(clean)) throw new Error(`Тема «${clean}» уже есть`);
    this.#topics.add(clean);
  }
}

function makeRow(student) {
  const tr = document.createElement("tr");
  const name = document.createElement("td");
  const marks = document.createElement("td");
  const average = document.createElement("td");
  const actions = document.createElement("td");

  name.textContent = student.name;
  marks.textContent = student.marks.join(", ");
  average.textContent = format(student.average);
  for (const mark of SETTINGS.marks) {

}

function submitWith(form, action) {
  const input = form.querySelector("input");
  form.addEventListener("submit", (event) => {
    event.preventDefault();
    try {
      action(input.value);
    } catch (e) {
      error.textContent = e.message;
      input.focus();
      return;
    }
    error.textContent = "";
    form.reset();
    render();
  });
}

// submitWith(form, (name) => group.add(name));   а не group.add
  1. Почему проверки ввода живут в классе, а не в слушателе формы?
    Ответ Тогда их нельзя обойти: форма, консоль и любой другой код добавляют ученика через тот же group.add. Слушатель только показывает текст ошибки.
  2. Зачем геттер marks возвращает копию массива?
    Ответ Иначе снаружи можно сделать push в сам приватный массив и обойти проверку addMark: средний получится больше пяти.
  3. Почему group.get("2") — undefined?
    Ответ Ключи Map — числа, а "2" — строка, другой ключ. Номер из dataset нужно превратить через Number.
  4. Что сломается, если написать yield* this.#students без .values()?
    Ответ Группа будет выдавать пары [номер, ученик], у пары нет name и marks, и makeRow упадёт.
  5. Почему submitWith(studentForm, group.add) не работает?
    Ответ Метод вызывается без объекта, this внутри — undefined (тело класса строгое). Нужна стрелка (name) => group.add(name) или bind.
